// ================== PINOS ==================
#define PINO_MOSFET 32     // Gate do IRLZ44N
#define MQ7_ANALOG  34     // A0 do MQ-7 via divisor 10k (em cima) / 20k (para o GND)
#define PINO_TESTE  25     // Botão KY-004 (INPUT_PULLUP)

// ================== CREDENCIAIS (gere novas!) ==================

#define LIMITE_MAX_PPM 50.0

// ================== HARDWARE DO MÓDULO ==================
const float VC_MODULO     = 5.0;      // tensão do OUT+ do LM2596 (meça com multímetro)
const float RL_MODULO     = 1000.0;   // resistor de carga do módulo em ohms. MEÇA!
                                      // (SMD "102" = 1k, "103" = 10k)
const float R_DIVISOR_TOTAL = 10000.0 + 20000.0;            // divisor fica em paralelo com o RL
const float RL_EFETIVO    = (RL_MODULO * R_DIVISOR_TOTAL) / (RL_MODULO + R_DIVISOR_TOTAL);
const float FATOR_DIVISOR = (10000.0 + 20000.0) / 20000.0;  // (Rtopo + Rbase) / Rbase
const float V_A0_MAX      = 4.6;      // acima disso o ADC do ESP32 satura (~3,07 V no pino)

// ---- Medição opcional do 5 V real (bateria descarregando derruba o LM2596) ----
// Hardware: OUT+ -> 20k -> GPIO35 -> 10k -> GND   (5 V vira ~1,67 V no pino)
#define MEDIR_VC 0                    // mude para 1 depois de montar esse divisor
#define VC_ANALOG 35
const float FATOR_VC = (20000.0 + 10000.0) / 10000.0;

// ================== CICLO DE AQUECIMENTO ==================
const unsigned long T_ALTO  = 60000;  // 60 s a 5 V (limpeza)
const unsigned long T_BAIXO = 90000;  // 90 s a 1,4 V (medição)
const unsigned long T_LEITURA = 89000;// lê no fim da fase baixa
const int DUTY_ALTO  = 255;
const int DUTY_BAIXO = 20;            // (1.4/5)^2 * 255 ≈ 20  → 1,4 V RMS

// ================== CURVA DO MQ-7 (Hanwei, CO) ==================
const float RATIO_AR_LIMPO = 27.5;    // Rs(ar limpo) / R0
const float CURVA_A = 99.042;
const float CURVA_B = -1.518;

// ================== CALIBRAÇÃO ==================
const int CICLOS_DESCARTE_CAL = 2;    // ciclos ignorados no início da calibração
const int AMOSTRAS_CAL        = 10;   // ciclos usados na média (~25 min)
const int CICLOS_AQUECIMENTO  = 4;    // sem alertas nos primeiros ciclos após ligar

WiFiClientSecure clientSSL;
WiFiClient clientTS;
UniversalTelegramBot bot(TELEGRAM_BOT_TOKEN, clientSSL);
Adafruit_BME280 bme;
WebServer server(80);
Preferences prefs;

// ================== VARIÁVEIS ==================
float ppm_final = 0, umidade_atual = 0, temp_atual = 0;
float v_a0 = 0, rs_atual = 0, R0 = 0;
String status_fase = "Iniciando...";
unsigned long tempoReferencia = 0, tempoBME = 0;
int fase = 1;
bool leituraFeita = false;
int ciclosCompletos = 0;

bool modoCalibracao = false;
double somaRsCal = 0;
int nCal = 0;

bool alertaEnviado = false, dispararAlerta = false, dispararNormalizado = false;
bool alertaSimulado = false;   // true quando o alerta veio do botão de teste
float ultimoPpmReal = 0;       // último valor medido de verdade (volta após o teste)

// ================== THINGSPEAK ==================
void enviarAoThingSpeak() {
  ThingSpeak.setField(1, ppm_final);
  ThingSpeak.setField(2, temp_atual);
  ThingSpeak.setField(3, umidade_atual);
  ThingSpeak.setField(4, rs_atual);   // diagnóstico (crie o Field 4 no canal)
  int codigo = ThingSpeak.writeFields(myChannelNumber, myWriteAPIKey);
  if (codigo == 200) Serial.println("[THINGSPEAK] Dados enviados com sucesso!");
  else Serial.println("[THINGSPEAK] Erro no envio. Código HTTP: " + String(codigo));
}

// ================== LEITURA DO MQ-7 ==================
// No módulo, aquecedor e circuito de medição dividem o mesmo VCC/GND.
// Por isso o MOSFET fica em 100% por ~20 ms durante a leitura: Vc fica estável
// em 5 V e a inércia térmica do sensor torna esse pulso desprezível.
float vc_atual = VC_MODULO;

float lerA0Volts() {
  analogWrite(PINO_MOSFET, DUTY_ALTO);
  delay(20);
  uint32_t soma = 0, somaVc = 0;
  for (int i = 0; i < 32; i++) {
    soma += analogReadMilliVolts(MQ7_ANALOG);   // já vem calibrado em mV
#if MEDIR_VC
    somaVc += analogReadMilliVolts(VC_ANALOG);
#endif
    delayMicroseconds(500);
  }
  analogWrite(PINO_MOSFET, DUTY_BAIXO);
#if MEDIR_VC
  vc_atual = (somaVc / 32.0) / 1000.0 * FATOR_VC;
  if (vc_atual < 4.75) Serial.printf("[ALERTA] 5V caiu para %.2f V - bateria baixa?\n", vc_atual);
#endif
  float v_pino = (soma / 32.0) / 1000.0;
  return v_pino * FATOR_DIVISOR;                // tensão real no A0 do módulo
}

float calcularRs(float v) {
  if (v < 0.01) return -1;
  return RL_EFETIVO * (vc_atual - v) / v;
}

float calcularPPM(float rs) {
  float ratio = rs / R0;
  return CURVA_A * pow(ratio, CURVA_B);
}

void processarLeitura() {
  v_a0 = lerA0Volts();
  rs_atual = calcularRs(v_a0);
  Serial.printf("\n[MQ7] V_A0=%.3f V | Rs=%.0f ohm | T=%.1f C | UR=%.1f%%\n",
                v_a0, rs_atual, temp_atual, umidade_atual);

  if (v_a0 < 0.02 || v_a0 > V_A0_MAX || rs_atual <= 0) {
    status_fase = "Erro: tensão do A0 fora da faixa";
    Serial.println("[MQ7] Leitura fora da faixa. Verifique fiação, divisor e RL.");
    return;
  }

  // ---- Modo calibração ----
  if (modoCalibracao) {
    if (ciclosCompletos < CICLOS_DESCARTE_CAL) {
      Serial.printf("[CAL] Descartando ciclo %d/%d\n", ciclosCompletos + 1, CICLOS_DESCARTE_CAL);
      status_fase = "Calibração: estabilizando";
      return;
    }
    somaRsCal += rs_atual;
    nCal++;
    status_fase = "Calibração " + String(nCal) + "/" + String(AMOSTRAS_CAL);
    Serial.printf("[CAL] Amostra %d/%d  Rs=%.0f ohm\n", nCal, AMOSTRAS_CAL, rs_atual);

    if (nCal >= AMOSTRAS_CAL) {
      float rsMedio = somaRsCal / nCal;
      R0 = rsMedio / RATIO_AR_LIMPO;
      prefs.putFloat("R0", R0);
      modoCalibracao = false;
      Serial.printf("[CAL] CONCLUÍDA! Rs(ar)=%.0f ohm -> R0=%.0f ohm (salvo na memória)\n", rsMedio, R0);
      status_fase = "Calibrado";
    }
    return;
  }

  // ---- Modo normal ----
  if (R0 <= 0) {
    status_fase = "Sem calibração (segure o botão ao ligar)";
    return;
  }

  ppm_final = calcularPPM(rs_atual);
  ultimoPpmReal = ppm_final;
  Serial.printf("[MQ7] Rs/R0=%.3f -> CO=%.1f ppm\n", rs_atual / R0, ppm_final);
  enviarAoThingSpeak();

  if (ciclosCompletos < CICLOS_AQUECIMENTO) {
    status_fase = "Aquecendo (alertas desativados)";
    return;
  }

  if (ppm_final >= LIMITE_MAX_PPM && !alertaEnviado) {
    dispararAlerta = true;
  } else if (ppm_final < (LIMITE_MAX_PPM - 5.0) && alertaEnviado && digitalRead(PINO_TESTE) == HIGH) {
    dispararNormalizado = true;
  }
}

// ================== SETUP ==================
void setup() {
  Serial.begin(115200);
  pinMode(PINO_MOSFET, OUTPUT);
  analogWrite(PINO_MOSFET, DUTY_ALTO);
  pinMode(PINO_TESTE, INPUT_PULLUP);

  // Faixa de 0 a ~3,1 V. Pino 34 é ADC1 e funciona com Wi-Fi ligado.
  analogSetPinAttenuation(MQ7_ANALOG, ADC_11db);
#if MEDIR_VC
  analogSetPinAttenuation(VC_ANALOG, ADC_11db);
#endif

  prefs.begin("mq7", false);
  R0 = prefs.getFloat("R0", 0);

  // Segure o botão por 3 s ao ligar para entrar no modo calibração
  if (digitalRead(PINO_TESTE) == LOW) {
    delay(3000);
    if (digitalRead(PINO_TESTE) == LOW) {
      modoCalibracao = true;
      Serial.println("[CAL] MODO CALIBRAÇÃO ATIVADO - deixe o sensor em AR LIMPO.");
    }
  }
  Serial.printf("R0 carregado da memória: %.0f ohm\n", R0);

  WiFiManager wm;
  wm.setConfigPortalTimeout(180);
  if (!wm.autoConnect("ESP32_Monitor_CO")) {
    Serial.println("Timeout na configuração. Reiniciando...");
    delay(3000);
    ESP.restart();
  }
  Serial.println("Conectado! IP: " + WiFi.localIP().toString());

  clientSSL.setInsecure();
  ThingSpeak.begin(clientTS);
  Wire.begin();
  if (!bme.begin(0x76)) Serial.println("Erro: BME280 não detectado!");
  else Serial.println("BME280 detectado com sucesso!");

  server.on("/", handleRoot);
  server.begin();

  fase = 1;
  status_fase = "Aquecendo Sensor (5V)";
  tempoReferencia = millis();
  tempoBME = millis();
}

// ================== LOOP ==================
void loop() {
  server.handleClient();
  unsigned long agora = millis();

  // BME280 a cada 2 s
  if (agora - tempoBME >= 2000) {
    float t = bme.readTemperature();
    float h = bme.readHumidity();
    if (!isnan(t) && !isnan(h) && h > 0.0) { temp_atual = t; umidade_atual = h; }
    else Wire.begin();
    tempoBME = agora;
  }

  // Ciclo do MQ-7
  if (fase == 1) {
    if (agora - tempoReferencia >= T_ALTO) {
      fase = 2;
      leituraFeita = false;
      analogWrite(PINO_MOSFET, DUTY_BAIXO);
      if (!modoCalibracao) status_fase = "Medindo (1.4V)";
      tempoReferencia = agora;
    }
  } else {
    if (!leituraFeita && agora - tempoReferencia >= T_LEITURA) {
      leituraFeita = true;
      processarLeitura();
    }
    if (agora - tempoReferencia >= T_BAIXO) {
      fase = 1;
      ciclosCompletos++;
      analogWrite(PINO_MOSFET, DUTY_ALTO);
      if (!modoCalibracao && R0 > 0 && ciclosCompletos >= CICLOS_AQUECIMENTO) status_fase = "Limpando sensor (5V)";
      tempoReferencia = agora;
    }
  }

  // Botão de teste (desativado durante a calibração)
  if (!modoCalibracao) {
    bool botaoApertado = (digitalRead(PINO_TESTE) == LOW);
    if (botaoApertado && !alertaSimulado) {
      // Início do teste: dispara UMA vez por aperto
      status_fase = "[MODO DE TESTE ATIVO]";
      ppm_final = 999.9;
      alertaSimulado = true;
      dispararAlerta = true;
      enviarAoThingSpeak();
    } else if (!botaoApertado && alertaSimulado) {
      // Botão solto: encerra a simulação e volta ao último valor medido
      static unsigned long debounceBotao = 0;
      if (millis() - debounceBotao > 1000) {
        debounceBotao = millis();
        ppm_final = ultimoPpmReal;
        if (!alertaEnviado) {
          // O alerta de teste não chegou a ser enviado: só limpa a simulação
          alertaSimulado = false;
          status_fase = "Teste encerrado (Telegram falhou)";
        } else if (ppm_final < (LIMITE_MAX_PPM - 5.0)) {
          status_fase = "Teste encerrado";
          dispararNormalizado = true;
        } else {
          alertaSimulado = false;   // CO real está alto: mantém o alerta como real
        }
      }
    }
  }

  // Telegram
  if (dispararAlerta) {
    dispararAlerta = false;
    String msg = alertaSimulado ? "⚠️ *ALERTA DE PERIGO (SIMULAÇÃO)!* ⚠️\n\n"
                                : "⚠️ *ALERTA DE PERIGO!* ⚠️\n\n";
    msg += "Níveis perigosos de Monóxido de Carbono (CO) detectados!\n";
    msg += "*Concentração:* " + String(ppm_final, 1) + " PPM\n";
    msg += "*Umidade:* " + String(umidade_atual, 1) + "%\n";
    msg += "*Temperatura:* " + String(temp_atual, 1) + "°C\n\n";
    msg += "Por favor, ventile o ambiente imediatamente!";
    if (bot.sendMessage(CHAT_ID, msg, "Markdown")) { Serial.println("[TELEGRAM] Alerta enviado!"); alertaEnviado = true; }
    else Serial.println("[TELEGRAM] Falha ao enviar alerta.");
  }

  if (dispararNormalizado) {
    dispararNormalizado = false;
    String msg = alertaSimulado ? "✅ *Nível de CO Normalizado (SIMULAÇÃO)* ✅\n\n"
                                : "✅ *Nível de CO Normalizado* ✅\n\n";
    msg += "*Concentração Atual:* " + String(ppm_final, 1) + " PPM";
    if (bot.sendMessage(CHAT_ID, msg, "Markdown")) {
      Serial.println("[TELEGRAM] Normalização enviada!");
      alertaEnviado = false;
      alertaSimulado = false;
    }
    else Serial.println("[TELEGRAM] Falha ao enviar normalização.");
  }
}
